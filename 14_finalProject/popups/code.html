int readTFLunaDistance() {
  int distance = -1;
  int count = 0;

  while (count < 50) {
    if (Serial1.available() >= 9) {
      if (Serial1.read() == 0x59 && Serial1.read() == 0x59) {
        uint8_t buffer[7];
        for (int i = 0; i < 7; i++) {
          buffer[i] = Serial1.read();
        }
        distance = buffer[0] + (buffer[1] << 8);
        count++;
      }
    }
  }
  return distance;
}
void sendSphericalToFirebase(float radius, int phiDeg, int thetaDeg) {
  // Convert degrees to radians
  float phi = phiDeg * PI / 180.0;
  float theta = thetaDeg * PI / 180.0;

  // Convert spherical to Cartesian
  float x = radius * sin(phi) * cos(theta);
  float y = radius * sin(phi) * sin(theta);
  float z = radius * cos(phi);

  // Create JSON with Cartesian coordinates
  String json = "{";
  json += "\"x\":" + String(x, 4) + ",";
  json += "\"y\":" + String(y, 4) + ",";
  json += "\"z\":" + String(z, 4);
  json += "}";

  Serial.print("Sending point: ");
  Serial.println(json);

  // Send to Firebase
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(firebaseBaseURL);
    http.addHeader("Content-Type", "application/json");

    int httpResponseCode = http.POST(json);
    Serial.print("HTTP Response code: ");
    Serial.println(httpResponseCode);

    if (httpResponseCode > 0) {
      String response = http.getString();
      Serial.println("Response: " + response);
    } else {
      Serial.println("Error sending POST");
    }

    http.end();
  } else {
    Serial.println("WiFi not connected");
  }
}

#include <WiFi.h>

const uint8_t slaveAddress = 0x08;
#define stepPin 13
#define dirPin 12

void setup() {
  pinMode(stepPin, OUTPUT);
  pinMode(dirPin, OUTPUT);
  Serial.begin(115200);
  Wire.begin();  // Master
}

void loop() {
  // 1. Move local stepper
  moveStepper(2);
  Serial.println("Master: Stepper moved");

  // 2. Send command to slave
  Wire.beginTransmission(slaveAddress);
  Wire.write(1);
  byte error = Wire.endTransmission();

  if (error == 0) {
    Serial.println("Master: Command sent to slave");

    // 3. Wait for confirmation
    byte confirmation = 0x00;
    while (confirmation != 0xAC) {
      Wire.requestFrom(slaveAddress, (uint8_t)1);
      if (Wire.available()) {
        confirmation = Wire.read();
        if (confirmation == 0xAC) {
          Serial.println("Master: Got confirmation from slave");
        } else {
          Serial.println("Master: Waiting...");
          delay(100);  // Wait and try again
        }
      }
    }
  } else {
    Serial.println("Master: Error sending data");
  }

  delay(1000);  // Optional pause between cycles
}

void moveStepper(int steps) {
  delay(50);
  if (steps == 0) return;

  digitalWrite(dirPin, steps > 0 ? HIGH : LOW);
  int totalSteps = abs(steps);
  for (int i = 0; i < totalSteps; i++) {
    digitalWrite(stepPin, HIGH);
    delay(5);
    digitalWrite(stepPin, LOW);
    delay(5);
  }
}

Explination

The process begins with the bottom microcontroller, an ESP32 Dev Module, which controls a Nema 17 stepper motor. The bottom microcontroller starts by stepping the motor once. After this step, it sends a signal, specifically, a 1, to the top microcontroller, an ESP32 Xiao, via I2C communication. When the top microcontroller receives this signal, it steps its motor 100 times, which corresponds to a total rotation of 180 degrees, before resetting. During each of these steps, the Xiao records both the theta and phi angles. At the same time, it takes distance measurements using a TF-Luna LiDAR sensor connected through serial. Each measurement gives a distance r, creating a point in polar coordinates (r,theta,phi) This point is then converted into Cartesian coordinates and sent to an external server (Firebase) for storage. Finally, the stored data is rendered live on a website I created, allowing visualization of the scanned points.
